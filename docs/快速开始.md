# GeoServer WMS/WMTS 快速开始指南

## 🚀 快速开始

### 方式一：简单直接（推荐新手）

在你的 `index-page.vue` 中：

```vue
<script setup>
import { onMounted } from 'vue';
import { 
  initCesium,
  viewerCesium,
  initWMSService,
  initWMTSService 
} from './js/threeCesium';

let wmsLayer = null;
let wmtsLayer = null;

onMounted(() => {
  // 1. 初始化 Cesium
  initCesium("cesiumContainer");
  
  // 2. 延迟加载图层（确保 viewer 初始化完成）
  setTimeout(() => {
    loadLayers();
  }, 2000);
});

function loadLayers() {
  // 加载 WMS 图层
  wmsLayer = initWMSService({
    url: "/geoserver/YG/wms",
    layers: "YG:HSQ_XZQ_MAP_BUFFER",
    parameters: {
      transparent: true,
      format: "image/png",
    },
    alpha: 0.8,
  });
  
  // 加载 WMTS 图层（如果有）
  wmtsLayer = initWMTSService({
    url: "/geoserver/gwc/service/wmts",
    layer: "YG:your_wmts_layer",
    style: "default",
    format: "image/png",
    alpha: 1.0,
  });
}

// 控制图层显示/隐藏
function toggleWMS() {
  if (wmsLayer) {
    wmsLayer.show = !wmsLayer.show;
  }
}
</script>
```

### 方式二：使用服务类（推荐进阶）

```vue
<script setup>
import { onMounted, ref } from 'vue';
import { initCesium, viewerCesium } from './js/threeCesium';
import { geoServerService } from '@/services/geoserverService';
import { GEOSERVER_CONFIG } from '@/config/geoserver.config';

const layerList = ref([]);

onMounted(async () => {
  // 初始化地图
  initCesium("cesiumContainer");
  
  // 测试连接
  const result = await geoServerService.testConnection();
  if (result.success) {
    console.log('✅ GeoServer 连接成功');
    setTimeout(() => loadAllLayers(), 2000);
  } else {
    console.error('❌ GeoServer 连接失败');
  }
});

function loadAllLayers() {
  const viewer = viewerCesium();
  const { layers } = GEOSERVER_CONFIG;
  
  // 加载所有 WMS 图层
  layers.wms.forEach(config => {
    if (config.visible) {
      geoServerService.loadWMSLayer(viewer, config);
    }
  });
  
  // 加载所有 WMTS 图层
  layers.wmts.forEach(config => {
    if (config.visible) {
      geoServerService.loadWMTSLayer(viewer, config);
    }
  });
  
  layerList.value = geoServerService.getAllLayers();
}

// 图层控制
function toggleLayer(layerId) {
  const layer = geoServerService.getLayer(layerId);
  if (layer) {
    geoServerService.setLayerVisible(layerId, !layer.layer.show);
  }
}

function setAlpha(layerId, alpha) {
  geoServerService.setLayerAlpha(layerId, alpha);
}
</script>

<template>
  <div class="layer-control">
    <div v-for="layer in layerList" :key="layer.id">
      <label>
        <input 
          type="checkbox" 
          :checked="layer.layer.show"
          @change="toggleLayer(layer.id)"
        />
        {{ layer.title }}
      </label>
      <input 
        type="range" 
        min="0" 
        max="1" 
        step="0.1"
        :value="layer.layer.alpha"
        @input="e => setAlpha(layer.id, parseFloat(e.target.value))"
      />
    </div>
  </div>
</template>
```

## 📝 配置你的图层

编辑 `src/config/geoserver.config.js`：

```javascript
export const GEOSERVER_CONFIG = {
  baseUrl: '/geoserver',  // 使用代理
  workspace: 'YG',
  
  layers: {
    wms: [
      {
        id: 'layer1',
        name: 'YG:你的图层名称',
        title: '图层显示名称',
        type: 'wms',
        visible: true,
        alpha: 0.8,
      }
    ],
    wmts: [
      {
        id: 'base',
        name: 'YG:底图图层',
        title: '底图',
        type: 'wmts',
        visible: true,
        alpha: 1.0,
      }
    ]
  }
};
```

## 🎨 常用操作

### 1. 控制图层显示/隐藏
```javascript
// 方式1：直接操作
wmsLayer.show = true/false;

// 方式2：使用服务类
geoServerService.setLayerVisible('layer_id', true/false);
```

### 2. 调整透明度
```javascript
// 方式1
wmsLayer.alpha = 0.5;

// 方式2
geoServerService.setLayerAlpha('layer_id', 0.5);
```

### 3. 调整图层顺序
```javascript
const viewer = viewerCesium();

// 移到最上层
viewer.imageryLayers.raiseToTop(wmsLayer);

// 移到最下层
viewer.imageryLayers.lowerToBottom(wmsLayer);
```

### 4. 移除图层
```javascript
// 方式1
viewer.imageryLayers.remove(wmsLayer);

// 方式2
geoServerService.removeLayer(viewer, 'layer_id');
```

## 🔧 故障排查

### 1. 图层不显示
```javascript
// 检查图层状态
console.log('显示:', wmsLayer.show);
console.log('透明度:', wmsLayer.alpha);
console.log('图层数量:', viewerCesium().imageryLayers.length);

// 强制显示
wmsLayer.show = true;
wmsLayer.alpha = 1.0;
viewerCesium().imageryLayers.raiseToTop(wmsLayer);
```

### 2. 测试 GeoServer 连接
```javascript
import { geoServerService } from '@/services/geoserverService';

const result = await geoServerService.testConnection();
console.log(result);
```

### 3. 查看所有图层
```javascript
const viewer = viewerCesium();
viewer.imageryLayers._layers.forEach((layer, index) => {
  console.log(`图层 ${index}:`, {
    show: layer.show,
    alpha: layer.alpha,
    provider: layer.imageryProvider.constructor.name
  });
});
```

## ⚡ 性能优化建议

1. **使用 WMTS 代替 WMS**（更快）
2. **启用 GeoWebCache 瓦片缓存**
3. **按需加载图层**（不要一次性加载所有）
4. **合理设置最大缩放级别**
